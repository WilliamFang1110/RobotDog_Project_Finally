// -------------------------- 按键中断处理 --------------------------
void ICACHE_RAM_ATTR handleButtonPress() {
    unsigned long currentTime = millis();
    if (currentTime - lastPressTime > debounceDelay) {
        buttonPressed = true;
        lastPressTime = currentTime;
    }
    unsigned long currentTime2 = millis();
    if (currentTime2 - lastPressTime2 > debounceDelay2) {
        buttonPressed2 = true;
        lastPressTime2 = currentTime2;
    }
}

// -------------------------- 百度定位获取 --------------------------
void fetchLocation() {
    if (initLocation || WiFi.status() != WL_CONNECTED || baiduApiKey.isEmpty()) {
        return;
    }
    WiFiClient client;
    HTTPClient http;
    String apiUrl = baiduApiUrl + baiduApiKey + "&coor=bd09ll";
    if (http.begin(client, apiUrl)) {
        int httpCode = http.GET();
        if (httpCode > 0) {
            String payload = http.getString();
            Serial.println("定位数据: " + payload);
            DynamicJsonDocument doc(2048);
            DeserializationError error = deserializeJson(doc, payload);
            if (!error) {
                int status = doc["status"];
                if (status == 0) {
                    JsonObject content = doc["content"];
                    JsonObject point = content["point"];
                    longitude = point["x"].as<String>();
                    latitude = point["y"].as<String>();
                    JsonObject addressDetail = content["address_detail"];
                    province = addressDetail["province"].as<String>();
                    city = addressDetail["city"].as<String>();
                    district = addressDetail["district"].as<String>();
                    cityname = city;
                    initLocation = true;
                    Serial.println("定位成功: " + province + city + district);
                    Serial.println("经纬度: " + latitude + "," + longitude);
                } else {
                    Serial.println("定位失败，错误码: " + String(status));
                }
            } else {
                Serial.println("定位JSON解析失败: " + String(error.c_str()));
            }
        } else {
            Serial.println("定位请求失败，状态码: " + String(httpCode));
        }
        http.end();
    }
}

// -------------------------- 心知天气获取 --------------------------
void fetchWeather() {
    if (WiFi.status() != WL_CONNECTED) {
        displayNoNet();
        return;
    }
    if (!initLocation) {
        do {
            u8g2.setFont(u8g2_font_ncenB08_tr);
            u8g2.drawStr(0, 32, "获取位置中...");
        } while (u8g2.nextPage());
        return;
    }
    if (!initweather && WiFi.status() == WL_CONNECTED && !cityname.isEmpty()) {
        WiFiClient client;
        HTTPClient http;
        String apiUrl = weatherAPI + weatherapi + "&location=" + cityname + 
                       "&language=zh-Hans&unit=c";
        if (http.begin(client, apiUrl)) {
            int httpCode = http.GET();
            if (httpCode > 0) {
                String payload = http.getString();
                DynamicJsonDocument doc(2048);
                DeserializationError error = deserializeJson(doc, payload);
                if (!error) {
                    temperature = doc["results"][0]["now"]["temperature"].as<String>();
                    weather = doc["results"][0]["now"]["text"].as<String>();
                    initweather = true;
                } 
            }
            http.end();
        }
    }
    do {
        u8g2.setFont(u8g2_font_ncenB08_tr);
        if (weather.indexOf("雨") != -1) u8g2.drawXBMP(0, 0, 64, 64, rain);
        else if (weather == "晴") u8g2.drawXBMP(0, 0, 64, 64, sun);
        else u8g2.drawXBMP(0, 0, 64, 64, cloud);
        u8g2.drawStr(64, 20, "Temp");
        u8g2.drawStr(64, 30, (temperature + " C").c_str());
    } while (u8g2.nextPage());
}

// -------------------------- 加载配置文件 --------------------------
void loadWiFiConfig() {
    if (SPIFFS.begin()) {
        if (SPIFFS.exists(ssidFile)) {
            fs::File file = SPIFFS.open(ssidFile, "r");
            if (file) {
                DynamicJsonDocument doc(1024);
                DeserializationError error = deserializeJson(doc, file);
                if (!error) {
                    String ssid = doc["ssid"].as<String>();
                    String pass = doc["pass"].as<String>();
                    weatherapi = doc["api"].as<String>();
                    baiduApiKey = doc["baiduapi"].as<String>();
                    WiFi.begin(ssid.c_str(), pass.c_str());
                    unsigned long start = millis();
                    while (WiFi.status() != WL_CONNECTED && millis() - start < 5000) {
                        delay(500);
                    }
                    if (WiFi.status() == WL_CONNECTED) {
                        Serial.println("WiFi已连接");
                        timeClient.begin();
                    } else {
                        Serial.println("WiFi连接失败，启动配置页");
                    }
                }
                file.close();
            }
        }
    }
}

// 前进动作：状态机推进
void frontNonBlock() {
    // 1. 动作完成：重置状态
    if (frontAction.currentStep >= sizeof(frontAction.steps)/sizeof(frontAction.steps[0])) {
        frontAction.currentStep = 0;
        frontAction.stepStartTime = 0;
        actionstate = 0; // 动作结束，重置动作状态
        return;
    }

    unsigned long currentTime = millis();
    int currentStepIdx = frontAction.currentStep;
    int stepDelay = frontAction.steps[currentStepIdx][4];

    // 2. 首次进入该步骤：设置舵机角度+记录开始时间
    if (frontAction.stepStartTime == 0) {
        servo1.write(frontAction.steps[currentStepIdx][0]);
        servo2.write(frontAction.steps[currentStepIdx][1]);
        servo3.write(frontAction.steps[currentStepIdx][2]);
        servo4.write(frontAction.steps[currentStepIdx][3]);
        frontAction.stepStartTime = currentTime; // 用millis()记录开始时间，替代delay()
        return;
    }

    // 3. 步骤延时到：推进到下一步
    if (currentTime - frontAction.stepStartTime >= stepDelay) {
        frontAction.currentStep++;
        frontAction.stepStartTime = 0; // 重置时间，为下一步做准备
    }
}

    }
}

// -------------------------- 电池检测函数 --------------------------
float getAverageAdcVoltage() {
    long total = 0;
    for (int i=0; i<numSamples; i++) {
        total += analogRead(A0);
        delay(10); // 电池采样延时不影响非阻塞动作（仅10ms，且主循环仍在推进）
    }
    return (total / (float)numSamples / 1023.0) * 1.0;
}
int mapBatteryPercentage(float voltage) {
    if (voltage <= minVoltage) return 0;
    if (voltage >= maxVoltage) return 100;
    return (int)((voltage - minVoltage) / (maxVoltage - minVoltage) * 100);
}

        Serial.println("收到指令: " + cmd);

        if (cmd == "front") {
            actionstate = 1;
            frontAction.currentStep = 0;
            frontAction.stepStartTime = 0;
        } else if (cmd == "back") {
            actionstate = 4;
            backAction.currentStep = 0;
            backAction.stepStartTime = 0;
        } else if (cmd == "left") {
            actionstate = 2;
            leftAction.currentStep = 0;
            leftAction.stepStartTime = 0;
        } else if (cmd == "right") {
            actionstate = 3;
            rightAction.currentStep = 0;
            rightAction.stepStartTime = 0;
        } else if (cmd == "toplefthand") {
            actionstate = 5;
            topLeftHandAction.currentStep = 0;
            topLeftHandAction.stepStartTime = 0;
        } else if (cmd == "toprighthand") {
            actionstate = 6;
            topRightHandAction.currentStep = 0;
            topRightHandAction.stepStartTime = 0;
        } else if (cmd == "sitdown") {
            actionstate = 8;
            sitdownAction.currentStep = 0;
            sitdownAction.stepStartTime = 0;
        } else if (cmd == "lie") {
            actionstate = 7;
            lieAction.currentStep = 0;
            lieAction.stepStartTime = 0;
        } else if (cmd == "dosleep") {
            actionstate = 10;
            doSleepAction.currentStep = 0;
            doSleepAction.stepStartTime = 0;
        } else if (cmd == "free") {
            freestate = true;
            freeModeWaitStart = millis();
        } else if (cmd == "offfree") {
            freestate = false;
        
        // 2. 表情/功能指令
        } else if (cmd == "kaixin") emojiState = 0;
        else if (cmd == "shengqi") emojiState = 1;
        else if (cmd == "cuowu") emojiState = 2;

// -------------------------- 主循环（非阻塞，millis()驱动） --------------------------
void loop() {
    // 1. 串口指令处理（优先：语音/手动指令）
    if (Serial.available() > 0) {
        serialListen();
    }
    
    // 2. 电池检测（周期性执行，不影响非阻塞动作）
    static unsigned long lastBatteryCheck = 0;
    if (millis() - lastBatteryCheck >= 1000) { // 1秒检测一次电池
        batteryVoltage = getAverageAdcVoltage() * voltageDividerRatio;
        batteryPercentage = mapBatteryPercentage(batteryVoltage);
        lastBatteryCheck = millis();
    }
    
    // 3. 按键处理（触发动作时重置状态机）
    if (buttonPressed) {
        buttonPressed = false;
        actionstate = 1; // 前进
        frontAction.currentStep = 0;
        frontAction.stepStartTime = 0;
    }
    if (buttonPressed2) {
        buttonPressed2 = false;
        actionstate = 4; // 后退
        backAction.currentStep = 0;
        backAction.stepStartTime = 0;
    }
    
    // 4. 联网状态更新（非阻塞：仅状态判断，无delay）
    if (WiFi.status() == WL_CONNECTED) {
        fetchLocation(); // 已定位则直接返回，不阻塞
        if (initLocation) {
            fetchWeather(); // 天气已获取则直接显示，不阻塞
        }
        timeClient.update(); // NTP更新非阻塞
    }
    
    // 5. 表情状态清屏（状态变化时执行一次）
    if (emojiState != prevEmojiState) {
        u8g2.clearDisplay();
        prevEmojiState = emojiState;
    }
    
    // 6. 自由模式（millis()实现非阻塞等待）
    if (freestate) {
        if (millis() - freeModeWaitStart >= freeModeWaitMs) {
            actionstate = random(1, 11); // 随机动作（1-10，跳过0）
            // 自由模式触发动作时，重置对应状态机（关键）
            switch(actionstate) {
                case 1: frontAction.currentStep=0; frontAction.stepStartTime=0; break;
                case 2: leftAction.currentStep=0; leftAction.stepStartTime=0; break;
                case 3: rightAction.currentStep=0; rightAction.stepStartTime=0; break;
                case 4: backAction.currentStep=0; backAction.stepStartTime=0; break;
                case 5: topLeftHandAction.currentStep=0; topLeftHandAction.stepStartTime=0; break;
                case 6: topRightHandAction.currentStep=0; topRightHandAction.stepStartTime=0; break;
                case 7: lieAction.currentStep=0; lieAction.stepStartTime=0; break;
                case 8: sitdownAction.currentStep=0; sitdownAction.stepStartTime=0; break;
                case 10: doSleepAction.currentStep=0; doSleepAction.stepStartTime=0; break;
                default: actionstate=0; break;
            }
            freeModeWaitStart = millis(); // 重置等待时间
        }
    }
    
    // 7. 执行非阻塞动作（状态机驱动，millis()判断步骤延时）
    switch (actionstate) {
        case 1: frontNonBlock(); break;
        case 2: leftNonBlock(); break;
        case 3: rightNonBlock(); break;
        case 4: backNonBlock(); break;
        case 5: topLeftHandNonBlock(); break;
        case 6: topRightHandNonBlock(); break;
        case 7: lieNonBlock(); break;
        case 8: sitdownNonBlock(); break;
        case 10: doSleepNonBlock(); break;
        case 9: emojiState = random(0,7); actionstate = 0; break; // 随机表情
        default: break;
    }
    
    // 8. OLED显示（非阻塞：仅绘制，无delay）
    switch (emojiState) {
        case 0: do { u8g2.drawXBMP(0,0,128,64,hi); } while(u8g2.nextPage()); break;
        case 1: do { u8g2.drawXBMP(0,0,128,64,angry); } while(u8g2.nextPage()); break;
        case 2: do { u8g2.drawXBMP(0,0,128,64,error); } while(u8g2.nextPage()); break;
        case 3: do { u8g2.drawXBMP(0,0,128,64,dowhat); } while(u8g2.nextPage()); break;
        case 4: do { u8g2.drawXBMP(0,0,128,64,love); } while(u8g2.nextPage()); break;
        case 5: do { u8g2.drawXBMP(0,0,128,64,sick); } while(u8g2.nextPage()); break;
        case 6: do { u8g2.drawXBMP(0,0,128,64,yun); } while(u8g2.nextPage()); break;
        case 7: fetchWeather(); break;
        case 8:
            if (WiFi.status() != WL_CONNECTED) {
                displayNoNet();
            } else {
                timeClient.update();
                time_t epochTime = timeClient.getEpochTime();
                struct tm *timeInfo = localtime(&epochTime);
                do {
                    u8g2.setFont(u8g2_font_ncenB08_tr);
                    u8g2.drawXBMP(0,0,64,64,timeimage);
                    String timeStr = String(timeInfo->tm_hour) + ":" + (timeInfo->tm_min<10?"0":"") + String(timeInfo->tm_min) + ":" + (timeInfo->tm_sec<10?"0":"") + String(timeInfo->tm_sec);
                    u8g2.drawStr(64,20,timeStr.c_str());
                    String dateStr = String(timeInfo->tm_year+1900) + "-" + (timeInfo->tm_mon+1<10?"0":"") + String(timeInfo->tm_mon+1) + "-" + (timeInfo->tm_mday<10?"0":"") + String(timeInfo->tm_mday);
                    u8g2.drawStr(64,35,dateStr.c_str());
                    String weekDays[] = {"Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"};
                    u8g2.drawStr(64,50,weekDays[timeInfo->tm_wday].c_str());
                } while(u8g2.nextPage());
            }
            break;
        case LOCATION_STATE:
            if (WiFi.status() != WL_CONNECTED) {
                displayNoNet();
            } else if (!initLocation) {
                do {
                    u8g2.setFont(u8g2_font_ncenB08_tr);
                    u8g2.drawStr(0, 32, "Getting location...");
                } while (u8g2.nextPage());
            } else {
                do {
                    u8g2.setFont(u8g2_font_ncenB08_tr);
                    u8g2.drawStr(0, 16, ("Pro: " + getProvinceEn(province)).c_str());
                    u8g2.drawStr(0, 32, ("City: " + getCityEn(city)).c_str());
                    u8g2.drawStr(0, 48, ("Lat: " + String(latitude.toFloat(), 2)).c_str());
                    u8g2.drawStr(0, 64, ("Lon: " + String(longitude.toFloat(), 2)).c_str());
                } while(u8g2.nextPage());
            }
            break;
    }
